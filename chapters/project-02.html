Expense Tracker

A CLI expense tracker with categories and spending summaries. Demonstrates multi-table design, GROUP BY, and SUM in a real program.

Project overview

Build a CLI expense tracker with categories, totals per category, and a monthly summary. Demonstrates GROUP BY, SUM, and multi-table design in a real program.

Skills used: Multiple tables · GROUP BY · SUM · AVG · date filtering · formatted output

Database Schema

CREATE TABLE categories (
    id   INTEGER PRIMARY KEY,
    name TEXT    NOT NULL UNIQUE
);

CREATE TABLE expenses (
    id          INTEGER PRIMARY KEY,
    amount      REAL    NOT NULL CHECK(amount > 0),
    description TEXT    NOT NULL,
    category_id INTEGER REFERENCES categories(id),
    date        TEXT    NOT NULL
);

Key Queries

-- Total spent per category
SELECT c.name, SUM(e.amount) AS total
FROM expenses e
JOIN categories c ON e.category_id = c.id
GROUP BY c.name
ORDER BY total DESC;

-- Monthly summary
SELECT
    SUBSTR(date, 1, 7) AS month,
    COUNT(*) AS transactions,
    SUM(amount) AS total,
    AVG(amount) AS avg_expense
FROM expenses
GROUP BY month
ORDER BY month DESC;

Complete Code

import sqlite3
from datetime import date

DB = "expenses.db"

def init_db():
    with sqlite3.connect(DB) as conn:
        conn.executescript("""
            CREATE TABLE IF NOT EXISTS categories (
                id INTEGER PRIMARY KEY, name TEXT NOT NULL UNIQUE);
            CREATE TABLE IF NOT EXISTS expenses (
                id INTEGER PRIMARY KEY,
                amount REAL NOT NULL CHECK(amount > 0),
                description TEXT NOT NULL,
                category_id INTEGER REFERENCES categories(id),
                date TEXT NOT NULL);
            INSERT OR IGNORE INTO categories (name)
            VALUES ('Food'),('Transport'),('Utilities'),('Entertainment'),('Other');
        """)

def add_expense(amount, description, category_name):
    with sqlite3.connect(DB) as conn:
        conn.row_factory = sqlite3.Row
        cat = conn.execute(
            "SELECT id FROM categories WHERE name=?", (category_name,)
        ).fetchone()
        if not cat:
            print(f"Category {category_name!r} not found."); return
        conn.execute(
            "INSERT INTO expenses (amount,description,category_id,date) VALUES (?,?,?,?)",
            (amount, description, cat["id"], date.today().isoformat())
        )

def summary_by_category():
    with sqlite3.connect(DB) as conn:
        conn.row_factory = sqlite3.Row
        rows = conn.execute("""
            SELECT c.name, SUM(e.amount) AS total, COUNT(*) AS count
            FROM expenses e JOIN categories c ON e.category_id=c.id
            GROUP BY c.name ORDER BY total DESC
        """).fetchall()
    print("\nSpending by category")
    print("-" * 35)
    for r in rows:
        print(f"{r['name']:<15} GHS {r['total']:>8.2f}  ({r['count']} items)")

def main():
    init_db()
    while True:
        print("\n1. Add expense  2. Summary  3. Quit")
        c = input("Choice: ").strip()
        if c == "1":
            amt  = float(input("Amount (GHS): "))
            desc = input("Description: ")
            cat  = input("Category (Food/Transport/Utilities/Entertainment/Other): ")
            add_expense(amt, desc, cat)
            print("Added.")
        elif c == "2": summary_by_category()
        elif c == "3": break

if __name__ == "__main__": main()
