CRUD with psycopg2

Full Create, Read, Update, Delete with psycopg2. You will see how closely it mirrors sqlite3 -- and exactly where it differs.

What you will learn
  • Full CRUD with psycopg2 using %s placeholders
  • Building a reusable database helper module
  • Handling psycopg2 exceptions
  • The complete task manager migrated to PostgreSQL

16.1 Full CRUD Functions

import psycopg2, os
from psycopg2.extras import RealDictCursor

def get_conn():
    return psycopg2.connect(
        dbname=os.environ["DB_NAME"], user=os.environ["DB_USER"],
        password=os.environ["DB_PASS"], host=os.environ.get("DB_HOST","localhost")
    )

def add_task(title, priority="medium"):
    with get_conn() as conn, conn.cursor() as cur:
        cur.execute(
            "INSERT INTO tasks (title, priority) VALUES (%s, %s) RETURNING id",
            (title, priority)
        )
        return cur.fetchone()[0]

def get_tasks(status=None):
    with get_conn() as conn, conn.cursor(cursor_factory=RealDictCursor) as cur:
        if status:
            cur.execute("SELECT * FROM tasks WHERE status=%s ORDER BY created_at DESC",(status,))
        else:
            cur.execute("SELECT * FROM tasks ORDER BY created_at DESC")
        return cur.fetchall()

def update_status(task_id, new_status):
    with get_conn() as conn, conn.cursor() as cur:
        cur.execute("UPDATE tasks SET status=%s WHERE id=%s",(new_status, task_id))

def delete_task(task_id):
    with get_conn() as conn, conn.cursor() as cur:
        cur.execute("DELETE FROM tasks WHERE id=%s",(task_id,))

16.2 Handling psycopg2 Exceptions

psycopg2.Error (base) psycopg2.IntegrityError UNIQUE / NOT NULL violated psycopg2.OperationalError connection failed, DB not running psycopg2.ProgrammingError bad SQL, wrong column name
try:
    task_id = add_task("Learn psycopg2")
except psycopg2.OperationalError:
    print("Cannot connect -- is PostgreSQL running?")
except psycopg2.IntegrityError:
    print("Constraint violated")
