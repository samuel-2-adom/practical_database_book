Notes App with Tags

Migrate your notes manager from JSON to SQLite -- including the many-to-many tag relationship. The most complete project in the book.

Project overview

Migrate your existing notes manager from JSON to SQLite -- including the many-to-many tag relationship. This is the most technically complete project in the book.

Skills used: Three tables · junction table · many-to-many joins · GROUP_CONCAT · full-text search · transactions

Database Schema

CREATE TABLE notes (
    id         INTEGER PRIMARY KEY,
    title      TEXT    NOT NULL,
    content    TEXT,
    created_at TEXT    NOT NULL,
    updated_at TEXT    NOT NULL
);

CREATE TABLE tags (
    id   INTEGER PRIMARY KEY,
    name TEXT    NOT NULL UNIQUE
);

CREATE TABLE note_tags (      -- junction table
    note_id INTEGER REFERENCES notes(id) ON DELETE CASCADE,
    tag_id  INTEGER REFERENCES tags(id),
    PRIMARY KEY(note_id, tag_id)
);

Key Queries

-- Note with all its tags as a comma-separated string
SELECT n.id, n.title, GROUP_CONCAT(t.name, ', ') AS tags
FROM notes n
LEFT JOIN note_tags nt ON n.id = nt.note_id
LEFT JOIN tags t       ON nt.tag_id = t.id
GROUP BY n.id
ORDER BY n.updated_at DESC;

-- Find all notes with a specific tag
SELECT n.title FROM notes n
JOIN note_tags nt ON n.id = nt.note_id
JOIN tags t       ON nt.tag_id = t.id
WHERE t.name = ?;

Core Functions

import sqlite3
from datetime import datetime

DB = "notes.db"

def add_note(title, content, tags=None):
    now = datetime.now().isoformat()
    with sqlite3.connect(DB) as conn:
        cur = conn.execute(
            "INSERT INTO notes (title,content,created_at,updated_at) VALUES (?,?,?,?)",
            (title, content, now, now)
        )
        note_id = cur.lastrowid
        if tags:
            _attach_tags(conn, note_id, tags)
    return note_id

def _attach_tags(conn, note_id, tag_names):
    for name in tag_names:
        conn.execute("INSERT OR IGNORE INTO tags (name) VALUES (?)", (name,))
        tag = conn.execute(
            "SELECT id FROM tags WHERE name=?", (name,)
        ).fetchone()
        conn.execute(
            "INSERT OR IGNORE INTO note_tags (note_id,tag_id) VALUES (?,?)",
            (note_id, tag[0])
        )

def find_by_tag(tag_name):
    with sqlite3.connect(DB) as conn:
        conn.row_factory = sqlite3.Row
        cur = conn.execute(
            "SELECT n.id, n.title, n.updated_at FROM notes n "
            "JOIN note_tags nt ON n.id=nt.note_id "
            "JOIN tags t ON nt.tag_id=t.id "
            "WHERE t.name=? ORDER BY n.updated_at DESC",
            (tag_name,)
        )
        return [dict(r) for r in cur.fetchall()]

def search_notes(query):
    with sqlite3.connect(DB) as conn:
        conn.row_factory = sqlite3.Row
        cur = conn.execute(
            "SELECT id, title FROM notes WHERE title LIKE ? OR content LIKE ?",
            (f"%{query}%", f"%{query}%")
        )
        return [dict(r) for r in cur.fetchall()]
