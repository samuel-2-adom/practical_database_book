CLI Contact Book

A complete command-line contact manager backed by SQLite. Add, search, and delete contacts with full error handling.

Project overview

Build a CLI contact book backed by SQLite. Add, search, update and delete contacts. Uses sqlite3, parameterised queries, and row factories. Single-file project you can run anywhere.

Skills used: sqlite3 · CREATE TABLE · INSERT · SELECT · UPDATE · DELETE · LIKE search · row factory · error handling

Database Schema

CREATE TABLE IF NOT EXISTS contacts (
    id      INTEGER PRIMARY KEY,
    name    TEXT    NOT NULL,
    email   TEXT    UNIQUE,
    phone   TEXT,
    notes   TEXT
);

Complete Code

import sqlite3

DB = "contacts.db"

def init_db():
    with sqlite3.connect(DB) as conn:
        conn.execute("CREATE TABLE IF NOT EXISTS contacts (
            id INTEGER PRIMARY KEY, name TEXT NOT NULL,
            email TEXT UNIQUE, phone TEXT, notes TEXT)")

def add_contact(name, email=None, phone=None, notes=None):
    try:
        with sqlite3.connect(DB) as conn:
            cur = conn.execute(
                "INSERT INTO contacts (name,email,phone,notes) VALUES (?,?,?,?)",
                (name, email, phone, notes)
            )
            return cur.lastrowid
    except sqlite3.IntegrityError:
        print("Email already exists.")
        return None

def search_contacts(query):
    with sqlite3.connect(DB) as conn:
        conn.row_factory = sqlite3.Row
        cur = conn.execute(
            "SELECT * FROM contacts WHERE name LIKE ? OR email LIKE ? ORDER BY name",
            (f"%{query}%", f"%{query}%")
        )
        return [dict(r) for r in cur.fetchall()]

def delete_contact(contact_id):
    with sqlite3.connect(DB) as conn:
        conn.execute("DELETE FROM contacts WHERE id=?", (contact_id,))

def main():
    init_db()
    while True:
        print("\n1. Add  2. Search  3. Delete  4. Quit")
        choice = input("Choice: ").strip()
        if choice == "1":
            name  = input("Name: ")
            email = input("Email (optional): ") or None
            phone = input("Phone (optional): ") or None
            cid = add_contact(name, email, phone)
            if cid: print(f"Added contact #{cid}")
        elif choice == "2":
            q = input("Search: ")
            results = search_contacts(q)
            for c in results:
                print(f"  [{c['id']}] {c['name']} | {c['email']} | {c['phone']}")
        elif choice == "3":
            cid = int(input("Contact ID to delete: "))
            delete_contact(cid); print("Deleted.")
        elif choice == "4": break

if __name__ == "__main__": main()
