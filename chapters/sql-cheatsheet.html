SQL Cheat Sheet

Every SQL statement and Python pattern from this book, in one place.

SELECT

SELECT col1, col2 FROM table;
SELECT * FROM table;
SELECT DISTINCT col FROM table;
SELECT col AS alias FROM table;

Filtering

WHERE col = 'value'
WHERE col != 'value'
WHERE col > 10
WHERE col BETWEEN 1 AND 100
WHERE col LIKE 'A%'       -- starts with A
WHERE col IN ('a', 'b')
WHERE col IS NULL
WHERE col IS NOT NULL
WHERE a AND b
WHERE a OR b
WHERE NOT a

Sorting and Limiting

ORDER BY col ASC
ORDER BY col DESC
ORDER BY col1 ASC, col2 DESC
LIMIT 10
LIMIT 10 OFFSET 20     -- page 3 of 10-per-page

Aggregates

SELECT COUNT(*) FROM table;
SELECT SUM(col) FROM table;
SELECT AVG(col) FROM table;
SELECT MIN(col), MAX(col) FROM table;

SELECT col, COUNT(*) FROM table GROUP BY col;
HAVING COUNT(*) > 5    -- filter after grouping

Joins

SELECT a.col, b.col
FROM table_a a
JOIN table_b b ON a.id = b.a_id;     -- INNER JOIN

LEFT JOIN table_b b ON a.id = b.a_id; -- all a, matched b or NULL

INSERT / UPDATE / DELETE

INSERT INTO table (col1, col2) VALUES ('a', 1);

UPDATE table SET col = 'value' WHERE id = 1;

DELETE FROM table WHERE id = 1;

Create / Drop

CREATE TABLE IF NOT EXISTS t (
    id      INTEGER PRIMARY KEY,          -- SQLite
    id      SERIAL  PRIMARY KEY,          -- PostgreSQL
    name    TEXT    NOT NULL,
    email   TEXT    UNIQUE,
    status  TEXT    DEFAULT 'active',
    age     INTEGER CHECK(age >= 0)
);

DROP TABLE IF EXISTS t;
ALTER TABLE t ADD COLUMN phone TEXT;

Python -- sqlite3

import sqlite3

with sqlite3.connect("db.sqlite3") as conn:
    conn.row_factory = sqlite3.Row
    cur = conn.execute("SELECT * FROM t WHERE id=?", (1,))
    row = cur.fetchone()       -- one row or None
    rows = cur.fetchall()     -- list of rows
    dict(row)                  -- convert to dict

    conn.execute("INSERT INTO t (name) VALUES (?)", (name,))
    -- ? placeholder, (value,) tuple, trailing comma for one value

Python -- psycopg2

import psycopg2
from psycopg2.extras import RealDictCursor

with psycopg2.connect(dbname="db", user="u", password="p") as conn:
    with conn.cursor(cursor_factory=RealDictCursor) as cur:
        cur.execute("SELECT * FROM t WHERE id=%s", (1,))
        row  = cur.fetchone()
        rows = cur.fetchall()

        cur.execute("INSERT INTO t (name) VALUES (%s) RETURNING id", (name,))
        new_id = cur.fetchone()["id"]
    -- %s placeholder  |  RETURNING for new row data
